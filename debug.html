📝 Firestore Rules yang Diperlukan

Salin rules ini ke Firebase Console → Firestore → Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // User yang login bisa baca/tulis dokumennya sendiri
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      // Admin bisa baca semua user
      allow read: if request.auth != null 
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
      allow write: if request.auth != null 
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
    // Koleksi role-spesifik
    match /admins/{userId}   { allow read, write: if request.auth != null; }
    match /ustadz/{userId}   { allow read, write: if request.auth != null; }
    match /santri/{userId}   { allow read, write: if request.auth != null; }
    match /wali_santri/{userId} { allow read, write: if request.auth != null; }
  }
}
